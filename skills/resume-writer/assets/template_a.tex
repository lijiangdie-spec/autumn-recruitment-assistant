% 区块标题（带下划线）
\newcommand{\rsection}[1]{\vspace{3pt}\noindent{\zihao{-4}\bfseries #1}\\[-0.85em]%
  \noindent{\color{rgray}\rule{\linewidth}{0.5pt}}\vspace{1pt}\par}
% 条目抬头：\rentry{左侧加粗}{右侧时间}
\newcommand{\rentry}[2]{\vspace{1pt}\noindent{\bfseries #1}\hfill{\color{rgray}#2}\par}
% 次行（GPA、技术栈等）
\newcommand{\rsub}[1]{\noindent #1\par}
% 页眉（带照片）：\rheader{姓名}{求职意向}{联系方式一行}{照片路径}
\newcommand{\rheader}[4]{%
  \noindent\begin{minipage}[c]{0.80\linewidth}
    {\zihao{2}\bfseries #1}\quad{\zihao{-4}\color{rgray}#2}\\[3pt]
    {\zihao{-5}\color{rgray}#3}
  \end{minipage}\hfill
  \begin{minipage}[c]{0.145\linewidth}\includegraphics[width=\linewidth]{#4}\end{minipage}\par}
% 页眉（无照片）：\rheadernophoto{姓名}{求职意向}{联系方式一行}
\newcommand{\rheadernophoto}[3]{%
  \noindent{\zihao{2}\bfseries #1}\quad{\zihao{-4}\color{rgray}#2}\\[3pt]
  {\zihao{-5}\color{rgray}#3}\par}
